\rel{Composition of tangles}
{\paset{scale=0.85}
\[
  \PA[numbers]{4,4}{D(2,2), A(1,1), id}\PA{4,4}{id, id, B(2,2)}\PA{4,2}{id, id, ev}
  \;\circ_3\;
  \PA{2,4}{coev, id, id}\PA{4,2}{id, ev, S(1,1)}
  \;=\;
  \PA[insert={3=T}]{4,4}{D(2,2), A(1,1), id}\PA{4,4}{id, id, B(2,2)}\PA{4,2}{id, id, ev}
  \;=\;
  \PA[insert={3=T}, reduce]{4,4}{D(2,2), A(1,1), id}\PA{4,4}{id, id, B(2,2)}\PA{4,2}{id, id, ev}
\]}

\rel{Loops and isotopy}
\[
  \PA{0,2}{coev}\PA{2,0}{ev} = \delta
  \quad
  \PA{1,3}{id, coev}\PA{3,1}{ev, id} = \PA{1,1}{id} = \PA{1,3}{coev, id}\PA{3,1}{id, ev}
  \quad
  \PA{2,4}{id, id, coev}\PA{4,2}{id, ev, id}\PA{2,2}{x}
  = \PA[reduce]{2,4}{id, id, coev}\PA{4,2}{id, ev, id}\PA{2,2}{x}
\]

\rel{The Temperley--Lieb relations}
\[
  e_1 = \PA{2,0}{ev}\PA{0,2}{coev}
  \qquad
  e_1^2 \;=\; \PA{2,0}{ev}\PA{0,2}{coev}\PA{2,0}{ev}\PA{0,2}{coev}
  \;=\; \PA[reduce]{2,0}{ev}\PA{0,2}{coev}\PA{2,0}{ev}\PA{0,2}{coev}
\]
\[
  e_1 e_2 e_1 \;=\; \PA[length=0.6]{3,1}{ev, id}\PA{1,3}{coev, id}\PA{3,1}{id, ev}\PA{1,3}{id, coev}
    \PA{3,1}{ev, id}\PA{1,3}{coev, id}
  \;=\; \PA[reduce]{3,1}{ev, id}\PA{1,3}{coev, id}\PA{3,1}{id, ev}\PA{1,3}{id, coev}
    \PA{3,1}{ev, id}\PA{1,3}{coev, id}
  \;=\; e_1
\]

\rel{Multiplication, trace and the conditional expectation}
\[
  x \cdot y = \PA{2,2}{x}\PA{2,2}{y}
  \qquad
  E(x) = \PA{1,3}{id, coev}\PA{3,3}{x(2,2), id}\PA{3,1}{id, ev}
  = \PA[reduce]{1,3}{id, coev}\PA{3,3}{x(2,2), id}\PA{3,1}{id, ev}
\]
\[
  \operatorname{tr}(x) = \PA{0,2}{coev}\PA{2,4}{id, coev, id}\PA{4,4}{x(2,2), id, id}
    \PA{4,2}{id, ev, id}\PA{2,0}{ev}
  = \PA[reduce]{0,2}{coev}\PA{2,4}{id, coev, id}\PA{4,4}{x(2,2), id, id}
    \PA{4,2}{id, ev, id}\PA{2,0}{ev}
\]

\rel{Sliding discs past each other}
\[
  \PA{2,2}{A(1,1), id}\PA{2,2}{id, B(1,1)}
  = \PA{2,2}{id, B(1,1)}\PA{2,2}{A(1,1), id}
  = \PA{2,2}{A(1,1), B(1,1)}
  = \PA[reduce]{2,2}{id, B(1,1)}\PA{2,2}{A(1,1), id}
\]
